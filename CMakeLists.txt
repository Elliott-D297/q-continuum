# WSLg's sound server reaches Windows over RDP and stalls longer than miniaudio's 3 x 10 ms low-latency buffer: the
# buffer runs dry and the cue stutters and repeats like an echo. 3 x 40 ms rides that out (a key beep lags ~0.1 s).
target_compile_definitions(raylib PRIVATE MA_DEFAULT_PERIOD_SIZE_IN_MILLISECONDS_LOW_LATENCY=40)

set(_assets "/* generated by CMake from assets/fonts -- do not edit */\n")
foreach(f ${QC_FONTS})
  get_filename_component(_name ${f} NAME)
  string(MAKE_C_IDENTIFIER "asset_${_name}" _sym)
  file(READ ${f} _hex HEX)
  string(LENGTH "${_hex}" _hl)
  math(EXPR _n "${_hl} / 2")
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _arr "${_hex}")
  string(APPEND _assets "const unsigned char ${_sym}[] = {${_arr}};\nconst unsigned int ${_sym}_len = ${_n};\n")
endforeach()
file(WRITE ${QC_ASSETS_C}.tmp "${_assets}")
configure_file(${QC_ASSETS_C}.tmp ${QC_ASSETS_C} COPYONLY)

# ---------------------------------------------------------------- warnings and sanitizers (our sources, not raylib)
option(QC_WERROR "Treat warnings in Q-Continuum's own sources as errors" ON)
option(QC_SANITIZE "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(QC_WARNINGS -Wall -Wextra -Wpedantic)
if(QC_WERROR)
  list(APPEND QC_WARNINGS -Werror)
endif()
function(qc_target t)
  target_compile_options(${t} PRIVATE ${QC_WARNINGS})
  if(QC_NATIVE)
    target_compile_options(${t} PRIVATE -march=native)
  endif()
  if(QC_SANITIZE)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
qc_target(qc_core)
qc_target(q-continuum)
